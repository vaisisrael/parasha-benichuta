נשא: מונולוג פקידותי בין הקלסרים

16.05.2026

-מתוך אוסף הספרייה הלאומית, מספר פריט: NNL_Ephemera997000957020405171-

ומלאה הארץ פקידים

מִבֶּן שְׁלֹשִׁים שָׁנָה וָמַעְלָה עַד בֶּן חֲמִשִּׁים שָׁנָה תִּפְקֹד אוֹתָם


הָעוֹלָם הוֹלֵךְ וּמִתְדַּרְדֵּר. זֶה בָּרוּר.

עֶרְכֵי יְסוֹד נֶעֱלָמִים בְּזֶה אַחַר זֶה - כְּמוֹ: וֶתֶק, גִּיל, וְכָתַב יָד עֲגַלְגַּל וְנָאֶה.

בְּרַאֲיוֹנוֹת הָעֲבוֹדָה, אִישׁ כְּבָר לֹא בּוֹדֵק אִם הַמֻּעֲמָד נֵחַן בְּרָמָה נְאוֹתָה שֶׁל סֵדֶר וְנִקָּיוֹן.

אַף אֶחָד לֹא מְבַקֵּשׁ מִמֶּנּוּ לְהַצִּיג אֶת הַמִּטְפַּחַת שֶׁבְּכִיסוֹ (זוֹ הַמְּיֹעֶדֶת לִמְחֹט אֶת חָטְמוֹ).

אֲפִלּוּ לֹא מְעִיפִים מַבָּט אֶל צִפָּרְנֵי יָדָיו, כְּדֵי לְוַדֵּא שֶׁהֵם גְּזוּרוֹת כַּהֲלָכָה!!!


פַּעַם? פַּעַם זֶה הָיָה אַחֶרֶת לְגַמְרֵי.

כְּדֵי לִהְיוֹת פָּקִיד, וְלוּ זוּטָר, אַפְרוּרִי, שָׁקוּף - הָיִיתָ צָרִיךְ לְהִוָּלֵד עִם קַו אִישִׁיּוּת שֶׁל סַרְגֵּל.

מִן שִׁלּוּב עָנֹג שֶׁל רַכּוּת וּנְחִישׁוּת. בַּעַל "רֹךְ פְּלָדָתִי".

נִדְרַשְׁתָּ לִכְתַב יָד עֲגַלְגַּל וְנָאֶה, אַךְ גַּם לִנְחִישׁוּת מִתְפָּרֶצֶת - לַהֲנָפַת הַחוֹתֶמֶת הַסַּמְכוּתִית וְהַטְבָּעָתָהּ הָרוֹעֶמֶת בַּמִּסְמָכִים.


"מִי יִתֵּן וְתִגְדַּל לִהְיוֹת פְּקִיד גְּבִיָּה, אוֹ לְמִצְעָר - פְּקִיד סַעַד", 

כָּךְ הָיְתָה לוֹחֶשֶׁת כָּל אֵם עִבְרִיָּה לִבְנָהּ הַקָּט, הַנִּרְדָּם.

פְּקִידִים מִצְטַיְּנִים - תֻּגְמְלוּ בְּטִיּוּל לְכִכַּר פִּיקָּדִילִי.

רַק הַקַּפְּדָנִים בֶּאֱמֶת, זָכוּ לָשֶׁבֶת מֵאֲחוֹרֵי הַדֶּלְפֵּק.

כָּל פְּקֻדָּה פְּקִידוּתִית, הָיְתָה אָז חֻקִּית בַּעֲלִיל.

אַף אֶחָד לֹא הֵעֵז לְהָנִיף דֶּגֶל שָׁחֹר מוּל הַמַּבָּט הַסַּמְכוּתִי שֶׁהִקְרִין פָּקִיד מִבַּעַד לְמִשְׁקְפֵי הַקֶּרֶן אוֹ הַמַּצְבֵּט.

וַעֲמַל הַפְּקִידִים -

הוֹ, עֲמַל הַפְּקִידִים.

תֻּגְמַל כַּהֲלָכָה בְּאֶמְצָעוּת 'עַמְלַת פָּקִיד'.


וְהַיּוֹם?

אֲבוֹי.

אֲפִלּוּ לְדַבֵּר עַל זֶה - צוֹבֵט לִי אֶת הַוִּילוֹנוֹת שֶׁל הַלֵּב.

כָּל צוּצִיק מִתְמַנֶּה לְפָקִיד.

וְעוֹד 'בָּכִיר'.

בְּלִי לַחְבֹּשׁ אֶת סַפְסְלֵי 'פִּטְמָן' - בֵּית הַסֵּפֶר הַגָּבוֹהַּ לִפְקִידִים (שָׁם לָמְדוּ לְשַׁלֵּב בֵּין סְטָנוֹגְרַפְיָה, סַבְלָנוּת וּשְׁתִיקָה רוֹעֶמֶת).


רַק נִפְקְדָה הָאֵם - וּכְבָר מְמַהֵר הָאָב לִפְתֹּחַ 'פִּקָּדוֹן לְכָל יֶלֶד'.

אֲנָשִׁים נוֹתְנִים פְּקֻדּוֹת קוֹלִיּוֹת לַטֵּלֵפוֹן!

צְעִירִים מַפְקִידִים בַּקְבּוּקִים בַּסּוּפֶּר, מְזֻמָּן בַּבַּנְק, נֶשֶׁק בַּכְּנִיסָה וְעַרְבוּת - בַּמִּשְׁטָרָה.

הַכֹּל מֻפְקָד - וְרַק הָעֲרָכִים? מִתְפַּזְּרִים.


הַכֹּל נִהְיָה מַמָּשׁ דִּיפּ - סְטֵיְט.

יְדַעְתֶּם שֶׁפְּקִידֵי בַּנְק מַצִּיעִים לְלָקוֹחוֹתֵיהֶם "פִּקָּדוֹן פָּסִיפְלוֹרָה'???


יֵשׁ לִי עוֹד הַרְבֵּה מָה לְהַגִּיד, אֲבָל אֲנִי לֹא רוֹצֶה לְהִשָּׁמַע זָקֵן טַרְחָן.

חוּץ מִזֶּה - 

אֲנִי צָרִיךְ לָצֵאת... וּלְהִתְפַּקֵּד לַלִּכּוּד.
